cmake_minimum_required(VERSION 3.20)
project(cpp_002 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(csv_parser src/csv_parser.cpp)
target_include_directories(csv_parser PUBLIC include)
enable_testing()
add_executable(csv_parser_visible tests/visible_tests.cpp)
target_link_libraries(csv_parser_visible PRIVATE csv_parser)
add_test(NAME csv_parser_visible COMMAND csv_parser_visible)
