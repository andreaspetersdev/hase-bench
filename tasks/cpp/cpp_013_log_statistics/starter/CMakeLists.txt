cmake_minimum_required(VERSION 3.20)
project(cpp_013 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(log_statistics src/log_statistics.cpp)
target_include_directories(log_statistics PUBLIC include)
enable_testing()
add_executable(log_statistics_visible tests/visible_tests.cpp)
target_link_libraries(log_statistics_visible PRIVATE log_statistics)
add_test(NAME log_statistics_visible COMMAND log_statistics_visible)
