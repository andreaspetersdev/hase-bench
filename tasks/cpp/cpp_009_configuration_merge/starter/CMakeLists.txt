cmake_minimum_required(VERSION 3.20)
project(cpp_009 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(config_value src/value.cpp)
target_include_directories(config_value PUBLIC include)
enable_testing()
add_executable(config_visible tests/visible_tests.cpp)
target_link_libraries(config_visible PRIVATE config_value)
add_test(NAME config_visible COMMAND config_visible)
