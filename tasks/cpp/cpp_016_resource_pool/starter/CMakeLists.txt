cmake_minimum_required(VERSION 3.20)
project(cpp_016 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(resource_pool src/resource_pool.cpp)
target_include_directories(resource_pool PUBLIC include)
enable_testing()
add_executable(resource_pool_visible tests/visible_tests.cpp)
target_link_libraries(resource_pool_visible PRIVATE resource_pool)
add_test(NAME resource_pool_visible COMMAND resource_pool_visible)
