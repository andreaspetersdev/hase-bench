cmake_minimum_required(VERSION 3.20)
project(cpp_019 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(message_recorder src/message_recorder.cpp)
target_include_directories(message_recorder PUBLIC include)
enable_testing()
add_executable(message_recorder_visible tests/visible_tests.cpp)
target_link_libraries(message_recorder_visible PRIVATE message_recorder)
add_test(NAME message_recorder_visible COMMAND message_recorder_visible)
set_tests_properties(message_recorder_visible PROPERTIES TIMEOUT 15)
