cmake_minimum_required(VERSION 3.20)
project(cpp_019_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(message_recorder message_recorder.cpp)
target_include_directories(message_recorder PUBLIC ../starter/include)
enable_testing()
add_executable(message_recorder_visible ../starter/tests/visible_tests.cpp)
target_link_libraries(message_recorder_visible PRIVATE message_recorder)
add_test(NAME message_recorder_visible COMMAND message_recorder_visible)
add_executable(message_recorder_hidden ../validator/hidden_tests.cpp)
target_link_libraries(message_recorder_hidden PRIVATE message_recorder)
add_test(NAME message_recorder_hidden COMMAND message_recorder_hidden)
set_tests_properties(message_recorder_visible message_recorder_hidden PROPERTIES TIMEOUT 30)
