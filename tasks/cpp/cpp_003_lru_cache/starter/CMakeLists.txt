cmake_minimum_required(VERSION 3.20)
project(cpp_003 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(lru INTERFACE)
target_include_directories(lru INTERFACE include)
enable_testing()

target_link_libraries(lru_visible PRIVATE lru)
add_test(NAME lru_visible COMMAND lru_visible)
