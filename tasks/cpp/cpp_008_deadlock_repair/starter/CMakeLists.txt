cmake_minimum_required(VERSION 3.20)
project(cpp_008 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ledger src/account.cpp)
target_include_directories(ledger PUBLIC include)
enable_testing()
add_executable(ledger_visible tests/visible_tests.cpp)
target_link_libraries(ledger_visible PRIVATE ledger)
add_test(NAME ledger_visible COMMAND ledger_visible)
set_tests_properties(ledger_visible PROPERTIES TIMEOUT 20)
