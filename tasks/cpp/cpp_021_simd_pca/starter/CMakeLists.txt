cmake_minimum_required(VERSION 3.20)
project(cpp_021 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pca src/pca.cpp src/avx2_covariance.cpp)
target_include_directories(pca PUBLIC include)
if(MSVC AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|i[3-6]86")
  set_source_files_properties(src/avx2_covariance.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|i[3-6]86")
  set_source_files_properties(src/avx2_covariance.cpp PROPERTIES COMPILE_OPTIONS "-mavx2")
endif()
enable_testing()
add_executable(pca_visible tests/visible_tests.cpp)
target_link_libraries(pca_visible PRIVATE pca)
add_test(NAME pca_visible COMMAND pca_visible)
