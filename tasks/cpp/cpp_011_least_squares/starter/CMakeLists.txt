cmake_minimum_required(VERSION 3.20)
project(cpp_011 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(least_squares src/least_squares.cpp)
target_include_directories(least_squares PUBLIC include)
enable_testing()
add_executable(least_squares_visible tests/visible_tests.cpp)
target_link_libraries(least_squares_visible PRIVATE least_squares)
add_test(NAME least_squares_visible COMMAND least_squares_visible)
