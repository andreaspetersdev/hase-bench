cmake_minimum_required(VERSION 3.20)
project(cpp_007 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(template_engine src/template.cpp src/template_cache.cpp)
target_include_directories(template_engine PUBLIC include)
enable_testing()
add_executable(template_visible tests/visible_tests.cpp)
target_link_libraries(template_visible PRIVATE template_engine)
add_test(NAME template_visible COMMAND template_visible)
set_tests_properties(template_visible PROPERTIES TIMEOUT 15)
