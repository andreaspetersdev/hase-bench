cmake_minimum_required(VERSION 3.20)
project(cpp_012 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(quaternion src/quaternion.cpp)
target_include_directories(quaternion PUBLIC include)
enable_testing()
add_executable(quaternion_visible tests/visible_tests.cpp)
target_link_libraries(quaternion_visible PRIVATE quaternion)
add_test(NAME quaternion_visible COMMAND quaternion_visible)
