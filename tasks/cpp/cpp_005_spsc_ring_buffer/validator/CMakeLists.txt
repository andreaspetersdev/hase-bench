cmake_minimum_required(VERSION 3.20)
project(cpp_005_hidden LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT STARTER_DIR)
  message(FATAL_ERROR "STARTER_DIR is required")
endif()

target_link_libraries(spsc_hidden PRIVATE spsc)

set_tests_properties(spsc_hidden PROPERTIES TIMEOUT 30)
