cmake_minimum_required(VERSION 3.20)
project(cpp_014 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(http_parser src/http_parser.cpp)
target_include_directories(http_parser PUBLIC include)
enable_testing()
add_executable(http_parser_visible tests/visible_tests.cpp)
target_link_libraries(http_parser_visible PRIVATE http_parser)
add_test(NAME http_parser_visible COMMAND http_parser_visible)
