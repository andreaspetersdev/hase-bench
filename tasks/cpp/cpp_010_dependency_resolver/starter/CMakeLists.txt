cmake_minimum_required(VERSION 3.20)
project(cpp_010 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dependency_resolver src/resolver.cpp)
target_include_directories(dependency_resolver PUBLIC include)
enable_testing()
add_executable(dependency_visible tests/visible_tests.cpp)
target_link_libraries(dependency_visible PRIVATE dependency_resolver)
add_test(NAME dependency_visible COMMAND dependency_visible)
