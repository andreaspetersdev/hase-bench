cmake_minimum_required(VERSION 3.20)
project(cpp_001 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(expression src/expression.cpp)
target_include_directories(expression PUBLIC include)
enable_testing()
add_executable(expression_visible tests/visible_tests.cpp)
target_link_libraries(expression_visible PRIVATE expression)
add_test(NAME expression_visible COMMAND expression_visible)
