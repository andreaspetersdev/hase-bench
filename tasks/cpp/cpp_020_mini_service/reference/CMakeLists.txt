cmake_minimum_required(VERSION 3.20)
project(cpp_020_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mini_service config.cpp parser.cpp stats.cpp work_queue.cpp service.cpp)
target_include_directories(mini_service PUBLIC ../starter/include)
enable_testing()
add_executable(mini_service_visible ../starter/tests/visible_tests.cpp)
target_link_libraries(mini_service_visible PRIVATE mini_service)
add_test(NAME mini_service_visible COMMAND mini_service_visible)
add_executable(mini_service_hidden_components ../validator/hidden_components.cpp)
target_link_libraries(mini_service_hidden_components PRIVATE mini_service)
add_test(NAME mini_service_hidden_components COMMAND mini_service_hidden_components)
add_executable(mini_service_hidden_integration ../validator/hidden_integration.cpp)
target_link_libraries(mini_service_hidden_integration PRIVATE mini_service)
add_test(NAME mini_service_hidden_integration COMMAND mini_service_hidden_integration)
set_tests_properties(mini_service_visible mini_service_hidden_components mini_service_hidden_integration PROPERTIES TIMEOUT 20)
