cmake_minimum_required(VERSION 3.20)
project(cpp_020_hidden LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT STARTER_DIR)
  message(FATAL_ERROR "STARTER_DIR is required")
endif()
add_subdirectory("${STARTER_DIR}" starter)
add_executable(mini_service_hidden_components hidden_components.cpp)
target_link_libraries(mini_service_hidden_components PRIVATE mini_service)
add_executable(mini_service_hidden_integration hidden_integration.cpp)
target_link_libraries(mini_service_hidden_integration PRIVATE mini_service)
enable_testing()
add_test(NAME mini_service_hidden_components COMMAND mini_service_hidden_components)
add_test(NAME mini_service_hidden_integration COMMAND mini_service_hidden_integration)
set_tests_properties(mini_service_hidden_components mini_service_hidden_integration PROPERTIES TIMEOUT 20)
