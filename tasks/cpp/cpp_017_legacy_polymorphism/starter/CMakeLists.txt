cmake_minimum_required(VERSION 3.20)
project(cpp_017 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(audit src/audit.cpp)
target_include_directories(audit PUBLIC include)
enable_testing()
add_executable(audit_visible tests/visible_tests.cpp)
target_link_libraries(audit_visible PRIVATE audit)
add_test(NAME audit_visible COMMAND audit_visible)
